cmake_minimum_required(VERSION 3.21)

include(FetchContent)
FetchContent_Declare(googletest
  GIT_REPOSITORY https://github.com/google/googletest.git
  GIT_TAG v1.14.0)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

set(PE_TESTS test_rc test_rlc test_switch test_buck test_netlist test_control test_thermal   test_converters test_magnetics test_adaptive test_recovery test_ac)

add_executable(test_rc test_rc.cpp)
add_executable(test_rlc test_rlc.cpp)
add_executable(test_switch test_switch.cpp)
add_executable(test_buck test_buck.cpp)
add_executable(test_netlist test_netlist.cpp)
add_executable(test_control test_control.cpp)
add_executable(test_thermal test_thermal.cpp)
add_executable(test_converters test_converters.cpp)
add_executable(test_magnetics test_magnetics.cpp)
add_executable(test_adaptive test_adaptive.cpp)
add_executable(test_recovery test_recovery.cpp)
add_executable(test_ac test_ac.cpp)

foreach(t IN LISTS PE_TESTS)
  target_link_libraries(${t} PRIVATE power_engine GTest::gtest_main)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /WX)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()

add_test(NAME RC_Transient COMMAND test_rc)
add_test(NAME RLC_Transient COMMAND test_rlc)
add_test(NAME SwitchDiode COMMAND test_switch)
add_test(NAME Buck_OpenLoop COMMAND test_buck)
add_test(NAME Netlist COMMAND test_netlist)
add_test(NAME Control COMMAND test_control)
add_test(NAME Thermal COMMAND test_thermal)
add_test(NAME Converters COMMAND test_converters)
add_test(NAME Magnetics COMMAND test_magnetics)
add_test(NAME Adaptive COMMAND test_adaptive)
add_test(NAME Recovery COMMAND test_recovery)
add_test(NAME AcAnalysis COMMAND test_ac)
